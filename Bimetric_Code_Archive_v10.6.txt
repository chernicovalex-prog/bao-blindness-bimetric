def nu_density(z, wnu_h):
    zp1 = 1.0 + z
    if zp1 <= Z_NR:

    return wnu_h * zp1**4 / Z_NR

# ============================================================
# Quartic solver (from §1, with WIDER brentq range)
# ============================================================
def quartic_coeffs(z, Om0, theta_rad, mFP2, sin2, cos2, alpha, beta, OL, Or):
    Ot = Om0 * (1+z)**3 + Or * (1+z)**4
    a0 = -1.0/3.0 * cos2 * mFP2 * (1 + 2*alpha + beta)
    a1 = (Ot + OL +
          mFP2 * (cos2*(alpha+beta) - sin2*(1 + alpha + beta/3.0)))
    a2 = mFP2 * (-cos2*beta + sin2*(1 + 2*alpha + beta))
    a3 = -(OL +
           1.0/3.0 * mFP2 * (cos2*(-1 + alpha - beta) + 3*sin2*(alpha + beta)))
    a4 = 1.0/3.0 * sin2 * mFP2 * beta
    return a0, a1, a2, a3, a4

def solve_y(z, Om0, theta_rad, mFP2, sin2, cos2, alpha, beta, OL, Or):
    # FIX-21: LCDM special case — mFP2=0 means no bimetric correction.
    # The quartic has no sign change on [1e-6, 1] (f>0 at both ends),
    # so brentq would fail and fallback with a warning for every z.
    if mFP2 == 0:
        return 1.0  # GR limit: y=1 → ODE=OL → LCDM
    a0, a1, a2, a3, a4 = quartic_coeffs(z, Om0, theta_rad, mFP2, sin2, cos2,
                                         alpha, beta, OL, Or)
    f = lambda y: a0 + a1*y + a2*y**2 + a3*y**3 + a4*y**4
    try:
        return brentq(f, 1e-6, 1.0, xtol=1e-14)
    except ValueError:
        import sys
        print(f"WARNING [§4a]: brentq fallback (y=1.0) at z={z}", file=sys.stderr)
        return 1.0  # GR limit fallback

# ============================================================
# Bimetric E^2(z) — FULL, not LCDM approximation
# ============================================================
def E2_bimetric(z, Om0, theta_rad, mFP2, sin2, cos2, alpha, beta, OL, Or, wnu_h=0.0):
    y = solve_y(z, Om0, theta_rad, mFP2, sin2, cos2, alpha, beta, OL, Or)
    dy = 1 - y
    ODE = OL - sin2 * mFP2 * dy * (1 + alpha*dy + beta/3.0 * dy**2)
    return Om0*(1+z)**3 + Or*(1+z)**4 + ODE + nu_density(z, wnu_h)

# ============================================================
# Self-consistent Omega_m from quartic at z=0
# ============================================================
def solve_Om0(theta_rad, mFP2, sin2, cos2, alpha, beta, OL, Or, wnu_h=0.0):
    def eq(Om):
        if Om <= 0 or Om >= 1:
            return 1e10
        y = solve_y(0.0, Om, theta_rad, mFP2, sin2, cos2, alpha, beta, OL, Or)
        dy = 1 - y
        ODE = OL - sin2 * mFP2 * dy * (1 + alpha*dy + beta/3.0*dy**2)
        return 1.0 - Or - ODE - Om - wnu_h
    return brentq(eq, 0.01, 0.99, xtol=1e-14)

# ============================================================
# Sound horizon r_s (direct integration of c_s/E(z))
# ============================================================
def sound_speed(z, wb):
    R_b = 31500 * wb * (T_CMB/2.7)**(-4) / (1+z)  # FIX-24
    cs = 1.0 / np.sqrt(3 * (1 + R_b))
    return cs

def r_s(z_rec, wb, h, Om0, OL_de, wnu_h=0.0):
    # Sound horizon integral using a-substitution.
    # At z > z_rec (~1090), Vainshtein screening is active, so
    # E^2(z) ≈ Om0*(1+z)^3 + Or*(1+z)^4 + OL_de (LCDM-like).
    # OL_de = effective dark energy density (≈ Omega_Lambda in screened regime).
    # FIX v10: added 1/a² Jacobian (dz = -da/a²) and E²(0)=1 normalization
    # to match D_M which uses E²_bimetric with E²(0)=1 by construction.
    def integrand(a):
        z = 1.0/a - 1.0
        cs = sound_speed(z, wb)
        E2 = Om0*(1+z)**3 + (omega_r_h2/h**2)*(1+z)**4 + OL_de + nu_density(z, wnu_h)
        # FIX-20: E²(0) = 1 by construction (solve_Om0 guarantees this).
        # OL_de is the INPUT parameter (0.699), NOT the true ODE(0).
        # For θ > 0, ODE(0) = OL - sin²θ·mFP²·dy0·(...) ≠ OL_de.
        # Using OL_de gave r_s overestimated by 0.17% (6.1°) to 4.9% (30°).
        E2_0 = 1.0
        return cs / (a*a * np.sqrt(E2 / E2_0))
    a_rec = 1.0 / (1.0 + z_rec)
    result, _ = quad(integrand, 1e-8, a_rec, limit=200)
    return result  # in units of c/H_0

# ============================================================
# D_M with bimetric E^2(z)
# ============================================================
def D_M_bimetric(z_rec, Om0, theta_rad, mFP2, sin2, cos2, alpha, beta, OL, Or,
                 wnu_h=0.0, npts=3000):
    # FIX v10.2: np.trapezoid → scipy.integrate.quad
    # Linear trapezoid with 3000 points misses the sharp bimetric transition
    # at z<10. Adaptive quad catches it. Effect on D_M: ~0.16%, but chi²
    # is hypersensitive to D_M (up to 15% shift in chi²).
    def integrand(z):
        E2 = E2_bimetric(z, Om0, theta_rad, mFP2, sin2, cos2,
                         alpha, beta, OL, Or, wnu_h)
        return 1.0 / np.sqrt(E2)
    result, _ = quad(integrand, 0, z_rec, limit=200)
    return result

# ============================================================
# CMB chi^2
# ============================================================
def chi2_cmb(vec):
    delta = vec - data_vec
    return float(delta @ C_inv @ delta)

def compute_observables(h, wb, Om0, OL, theta_rad, mFP2, sin2, cos2,
                        alpha, beta, Or, wnu=omega_nu, wnu_h=0.0):
    DM = D_M_bimetric(z_star, Om0, theta_rad, mFP2, sin2, cos2,
                       alpha, beta, OL, Or, wnu_h)
    R = np.sqrt(Om0 + wnu_h) * DM
    rs = r_s(z_star, wb, h, Om0, OL, wnu_h)
    # l_a
    l_a = np.pi * DM / rs
    return np.array([R, l_a, wb])

def minimize_chi2(Om0, OL, theta_rad, mFP2, sin2, cos2, alpha, beta, Or, wnu_h=0.0):
    def objective(params):
        h, wb = params
        if not (0.55 < h < 0.88 and 0.015 < wb < 0.025):  # FIX-23 wide h
            return 1e10
        vec = compute_observables(h, wb, Om0, OL, theta_rad, mFP2, sin2, cos2,
                                  alpha, beta, Or, wnu_h=wnu_h)
        return chi2_cmb(vec)
    # Coarse scan (FIX v10.2: wb range expanded to [0.015, 0.025])
    best = 1e10; best_init = None
    for h in np.linspace(0.60, 0.85, 16):  # FIX-23 wide coarse grid
        for wb in np.linspace(0.015, 0.025, 12):
            c2 = objective([h, wb])
            if c2 < best:

    res = minimize(objective, best_init, method='Nelder-Mead',
                   options={'xatol': 1e-8, 'fatol': 1e-12, 'maxiter': 5000})
    h_best, wb_best = res.x
    vec = compute_observables(h_best, wb_best, Om0, OL, theta_rad, mFP2,
                               sin2, cos2, alpha, beta, Or, wnu_h=wnu_h)
    return chi2_cmb(vec), h_best, wb_best, vec

# ============================================================
# Quartic Solver (Hogas 2025, Appendix A, eq. A13)
# ============================================================
def solve_quartic(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0=0.3044):
    """Solve quartic for y(z) using Brent's method on [1e-6, 1.0].

    Quartic (A13): a0 + a1*y + a2*y^2 + a3*y^3 + a4*y^4 = 0
    where y = a_f / a_g (ratio of scale factors).
    """
    Omega_tot_z = Omega_m0 * (1+z)**3 + Omega_r * (1+z)**4

    a0 = -1.0/3.0 * cos2 * mFP2 * (1 + 2*alpha + beta)
    a1 = (Omega_tot_z + Omega_Lambda +
          mFP2 * (cos2*(alpha+beta) - sin2*(1 + alpha + beta/3.0)))
    a2 = mFP2 * (-cos2*beta + sin2*(1 + 2*alpha + beta))
    a3 = -(Omega_Lambda +
           1.0/3.0 * mFP2 * (cos2*(-1 + alpha - beta) + 3*sin2*(alpha + beta)))
    a4 = 1.0/3.0 * sin2 * mFP2 * beta

    def quartic_eval(y):
        return a0 + a1*y + a2*y**2 + a3*y**3 + a4*y**4

    try:
        return brentq(quartic_eval, 1e-6, 1.0)
    except ValueError:
        import sys
        print(f"WARNING [§1]: brentq fallback (y=1.0) at z={z}", file=sys.stderr)
        return 1.0  # GR limit fallback


def E2_bimetric(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0=0.3044):
    """Compute E^2(z) = H^2(z)/H_0^2 for bimetric model (A11 + A12)."""
    y = solve_quartic(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0)
    dy = 1 - y
    Omega_DE = Omega_Lambda - sin2 * mFP2 * dy * (1 + alpha*dy + beta/3.0 * dy**2)
    return Omega_m0*(1+z)**3 + Omega_r*(1+z)**4 + Omega_DE


def w_DE(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0=0.3044, dz=1e-4):
    """Effective equation of state w_DE(z).

    Formula: w = -1 + (1+z)/(3*Omega_DE) * d(Omega_DE)/dz

    Equivalent to Hogas 2021a eq. (5.9):
      w = -1 - (1/3) * y' * d(ln Omega_DE)/dy
    where y' = dy/d(ln a) = -(1+z) * dy/dz (negative).
    The minus signs cancel, giving the PLUS sign in the code formula.
    """
    yp = solve_quartic(z + dz, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0)
    y0 = solve_quartic(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0)

    def ODE(y):
        d = 1 - y
        return Omega_Lambda - sin2 * mFP2 * d * (1 + alpha*d + beta/3.0 * d**2)

    # FIX v10.1: one-sided difference at z=0 (avoids z-dz < 0)
    if z < dz:
        ym = y0  # no backward step
        dODE_dz = (ODE(yp) - ODE(y0)) / dz
    else:
        ym = solve_quartic(z - dz, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0)
        dODE_dz = (ODE(yp) - ODE(ym)) / (2 * dz)
    return -1.0 + (1+z) / (3.0 * ODE(y0)) * dODE_dz


# ============================================================
# BAO Distance Predictions
# ============================================================
def _integrate(y, x):
    return np.sum(0.5 * (y[:-1] + y[1:]) * (x[1:] - x[:-1]))


def comoving_distance(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0, Nz=500):
    """D_M = c/H_0 * integral(1/E(z'), dz')."""
    zs = np.linspace(0, z, Nz)
    E2s = np.array([E2_bimetric(zi, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0)
                    for zi in zs])
    return _integrate(1.0 / np.sqrt(E2s), zs)


def BAO_predictions(f, z, obs_type, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0):
    """Compute BAO observable: D_M/r_d, D_H/r_d, or D_V/r_d.

    f = c / (H_0 * r_d)  -- free parameter marginalised over.
    D_V = (z * D_M^2 * D_H)^{1/3}  (Eisenstein & Hu, used by DESI)
    """
    if obs_type == 'DH':
        return f / np.sqrt(E2_bimetric(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0))
    elif obs_type == 'DM':
        return f * comoving_distance(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0)
    elif obs_type == 'DV':
        E2 = E2_bimetric(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0)
        DM = comoving_distance(z, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0)
        DH = 1.0 / np.sqrt(E2)
        return f * (z * DM**2 * DH)**(1.0/3.0)


# ============================================================
# Chi-squared Calculation
# ============================================================
def chi2_bimetric(f, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0):
    model = np.array([BAO_predictions(f, z, ot, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0)
                      for z, ot, v, e, r in desi_data])
    delta = data_vals - model
    return float(delta @ C_inv @ delta)


def minimize_chi2_bimetric(mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0):
    result = minimize_scalar(
        lambda f: chi2_bimetric(f, mFP2, sin2, cos2, alpha, beta, Omega_Lambda, Omega_r, Omega_m0),
        bounds=(20, 80), method='bounded')
    return result.x, result.fun


def chi2_LCDM(f, Omega_m0, Omega_r=5.38e-5):
    model = np.zeros(n)
    for i, (z, ot, v, e, r) in enumerate(desi_data):
        E2 = Omega_m0*(1+z)**3 + Omega_r*(1+z)**4 + (1 - Omega_m0 - Omega_r)
        if ot == 'DH':
            model[i] = f / np.sqrt(E2)
        elif ot == 'DM':
            zs = np.linspace(0, z, 500)
            E2s = Omega_m0*(1+zs)**3 + Omega_r*(1+zs)**4 + (1 - Omega_m0 - Omega_r)
            model[i] = f * _integrate(1.0/np.sqrt(E2s), zs)
        elif ot == 'DV':
            zs = np.linspace(0, z, 500)
            E2s = Omega_m0*(1+zs)**3 + Omega_r*(1+zs)**4 + (1 - Omega_m0 - Omega_r)
            DM = _integrate(1.0/np.sqrt(E2s), zs)
            DH = 1.0 / np.sqrt(E2)
            model[i] = f * (z * DM**2 * DH)**(1.0/3.0)
    delta = data_vals - model
    return float(delta @ C_inv @ delta)


def minimize_chi2_LCDM(Omega_m0, Omega_r=5.38e-5):
    result = minimize_scalar(lambda f: chi2_LCDM(f, Omega_m0, Omega_r),
                             bounds=(20, 80), method='bounded')
    return result.x, result.fun


# ============================================================
# B-parameter conversion (Hogas 2021a, eq. 3.4a-e)
# ============================================================
def compute_B_params(theta_deg, mFP_H0, alpha, beta, Omega_Lambda):
    """Convert physical parameters to B-parameters (forward, eq. 3.4a-e)."""
    th = np.radians(theta_deg)
    s2 = np.sin(th)**2
    m2 = mFP_H0**2

    B0 = 3*Omega_Lambda - s2*m2*(3 + 3*alpha + beta)                     # 3.4a
    B1 = s2*m2*(1 + 2*alpha + beta)                                       # 3.4b
    B2 = -s2*m2*(alpha + beta)                                            # 3.4c
    B3 = s2*m2*beta                                                       # 3.4d
    B4 = 3*np.tan(th)**2 * Omega_Lambda + s2*m2*(-1 + alpha - beta)       # 3.4e

    return {'B0': B0, 'B1': B1, 'B2': B2, 'B3': B3, 'B4': B4}


def invert_B_params(B, theta_deg):
    """Convert B-parameters to physical parameters (inverse, eq. 3.3a-e)."""
    th = np.radians(theta_deg)
    s2 = np.sin(th)**2
    B0, B1, B2, B3, B4 = B['B0'], B['B1'], B['B2'], B['B3'], B['B4']

    mFP2 = (B1 + 2*B2 + B3) / s2                                          # 3.3b
    Omega_Lambda = B0/3 + B1 + B2 + B3/3                                  # 3.3c
    alpha = -(B2 + B3) / (B1 + 2*B2 + B3)                                # 3.3d
    beta = B3 / (B1 + 2*B2 + B3)                                         # 3.3e

    return {'mFP2': mFP2, 'Omega_Lambda': Omega_Lambda, 'alpha': alpha, 'beta': beta}

#!/usr/bin/env python3
"""
SNe Delta-chi2: v5
- Correct quartic (y(0) = 0.999512, Omega_m = 0.30435)
- m_b_corr_err_DIAG for diagonal covariance (matches article: 15.58)
- Full STAT+SYS for full covariance (VERIFIED rev 38: 6.50/0.96/0.3470)
- Optional STATONLY for comparison
- sigma_int = 0.0
- zCMB for integral, zHEL for (1+z) factor
- Analytical marginalization over M
"""

import numpy as np
import pandas as pd
from scipy.optimize import brentq
from scipy.integrate import cumulative_trapezoid
from scipy.interpolate import interp1d
import os, time

# ===== MODEL PARAMETERS =====
theta = 6.1 * np.pi / 180
m_FP, alpha, beta = 25.0, 0.0, 3.0
Omega_Lambda, Omega_r = 0.699, 9.08e-5
sin2, tan2, m2 = np.sin(theta)**2, np.tan(theta)**2, m_FP**2

# B-parameters (Eq. 3.4, Hogas & Mortsell 2021)
B0 = 3*Omega_Lambda - sin2*m2*(3 + 3*alpha + beta)
B1 = sin2*m2*(1 + 2*alpha + beta)
B2 = -sin2*m2*(alpha + beta)
B3 = sin2*m2*beta
B4 = 3*tan2*Omega_Lambda + sin2*m2*(-1 + alpha - beta)

print("="*60)
print("SNe Delta-chi2: v5")
print("  m_b_corr_err_DIAG (diagonal) + full STAT+SYS")
print("  sigma_int = 0.0, zCMB integral, zHEL (1+z)")
print("="*60)
print(f"\nB-params: B0={B0:.3f} B1={B1:.3f} B2={B2:.3f} B3={B3:.3f} B4={B4:.3f}")
print(f"  (article: -40.248, 28.230, -21.173, 21.173, -28.206)")

# ===== QUARTIC SOLVER =====
def solve_y(z, Om):
    Ot = Om*(1+z)**3 + Omega_r*(1+z)**4
    a0 = -B1/(3*tan2)
    a1 = B0/3 - B2/tan2 + Ot
    a2 = B1 - B3/tan2
    a3 = B2 - B4/(3*tan2)
    a4 = B3/3
    return brentq(lambda y: a0 + a1*y + a2*y**2 + a3*y**3 + a4*y**4, 1e-6, 1.0)

def E2_bim(z, Om, y):
    Ot = Om*(1+z)**3 + Omega_r*(1+z)**4
    ODE = Omega_Lambda - sin2*m2*(1-y)*(1 + alpha*(1-y) + beta/3*(1-y)**2)
    return Ot + ODE

def E2_lcdm(z, Om):
    return Om*(1+z)**3 + Omega_r*(1+z)**4 + (1 - Om - Omega_r)

# ===== SELF-CONSISTENCY =====
Omega_m = brentq(lambda Om: E2_bim(0, Om, solve_y(0, Om)) - 1.0, 0.01, 0.99)
y0 = solve_y(0, Omega_m)
print(f"\nSelf-consistency:")
print(f"  Omega_m = {Omega_m:.5f}  (article: 0.30439)")
print(f"  y(0) = {y0:.6f}  (article: 0.999512)")
print(f"  E^2(0) = {E2_bim(0, Omega_m, y0):.8f}")

# ===== BIMETRIC E(z) ON FINE GRID =====
c_H0 = 2997.92458  # c/H0 in Mpc (h=1)
z_fine = np.linspace(0, 5, 10000)
t0 = time.time()
y_fine = np.array([solve_y(z, Omega_m) for z in z_fine])
E_fine_bim = np.sqrt(E2_bim(z_fine, Omega_m, y_fine))
int_fine_bim = cumulative_trapezoid(1.0/E_fine_bim, z_fine, initial=0)
int_bim_interp = interp1d(z_fine, int_fine_bim, kind='cubic', fill_value='extrapolate')
print(f"\nBimetric E(z) grid: {time.time()-t0:.1f}s")

# ===== FIND FILES =====
print(f"\n1. Finding files...")
data_file = cov_file = cov_statonly = None
for f in os.listdir('.'):
    fl = f.lower()
    if f.endswith('.dat') and 'sh0es' in fl and 'cov' not in fl:
        data_file = f
    elif f.endswith('.cov') and 'stat+sys' in fl.replace('%2b','+'):
        cov_file = f
    elif f.endswith('.cov') and 'statonly' in fl:
        cov_statonly = f

if cov_statonly:
    print(f"  Stat-only: {cov_statonly}")

print(f"  Columns: {list(df.columns[:10])}...")
zCMB = df['zCMB'].values

mb = df['m_b_corr'].values
mb_err = df['m_b_corr_err_DIAG'].values
print(f"  Total SNe: {len(zCMB)}")

# ===== READ COVARIANCE(S) =====
print(f"\n3. Reading covariance...")
t0 = time.time()
with open(cov_file) as f:

    vals = [float(next(f).strip()) for _ in range(N_cov * N_cov)]

print(f"  diag check: sqrt(diag(STAT+SYS))/m_b_corr_err_DIAG = {np.mean(np.sqrt(np.diag(cov_full))/mb_err):.4f}")

cov_statonly_mat = None
if cov_statonly:

print(f"\n4. Applying zCMB > 0.01 cut...")
mask = zCMB > 0.01

def chi2_diag(mu):
    r = mb_c - mu
    w = 1.0 / err_c**2
    S = np.sum(w)
    return np.sum(w * r**2) - np.sum(w * r)**2 / S

def chi2_full(mu, Cinv, S_full):
    r = mb_c - mu
    Cinv_r = Cinv @ r
    return r @ Cinv_r - np.sum(Cinv_r)**2 / S_full

print(f"  500 Omega_m values x {N} SNe: {time.time()-t0:.1f}s")

# LCDM at fixed Omega_m (bimetric prediction)
mu_lcdm_fixed = mu_lcdm_grid[np.argmin(np.abs(Om_grid - Omega_m))]

# ===== DIAGONAL COVARIANCE =====
print(f"\n7. Diagonal (m_b_corr_err_DIAG, sigma_int=0.0)...")
t0 = time.time()
chi2_diag_grid = np.array([chi2_diag(mu) for mu in mu_lcdm_grid])

for Om in Om_fine_d:

chi2_fine_d = np.array(chi2_fine_d)
i_best_d2 = np.argmin(chi2_fine_d)
Om_best_d, chi2_best_d = Om_fine_d[i_best_d2], chi2_fine_d[i_best_d2]

chi2_bim_d = chi2_diag(mu_bim)
# LCDM at fixed Omega_m
mu_fixed_d = 5*np.log10((1+zH)*c_H0*np.interp(zC, z_fine, 
    cumulative_trapezoid(1.0/np.sqrt(E2_lcdm(z_fine, Omega_m)), z_fine, initial=0))) + 25
chi2_fixed_d = chi2_diag(mu_fixed_d)

dchi2_std_d = chi2_bim_d - chi2_best_d
dchi2_pure_d = chi2_bim_d - chi2_fixed_d
print(f"  LCDM: Om={Om_best_d:.4f}, chi2={chi2_best_d:.4f}  ({time.time()-t0:.1f}s)")

print(f"  chi2/dof = {chi2_best_d/(N-1):.3f}")

t0 = time.time()
Cinv = np.linalg.inv(cov_c)
S_full = np.sum(Cinv)
print(f"  Inverted in {time.time()-t0:.1f}s")

chi2_fine_f = np.array(chi2_fine_f)

"""
SNe Delta-chi2: v6 — sigma_int sensitivity test
Based on v5, with one change: sigma_int added to diagonal covariance.

Changes from v5:
  - w = 1.0 / err_c**2            -->  w = 1.0 / (err_c**2 + sigma_int**2)

Data source: Pantheon+SH0ES.dat (GitHub: PantheonPlusSH0ES/DataRelease)

  (rev 38: combined values must use one consistent covariance scheme;
  mixing full-cov BAO with diag SNe is not interpretable)

Conclusion: SNe remain discriminators even with sigma_int=0.12.
Article conclusion unchanged. V6 can be cited as conservative estimate.

Author: Alisa AI
Date: 2026-10-07
"""

import numpy as np
from scipy.optimize import brentq
from scipy.integrate import cumulative_trapezoid
from scipy.interpolate import interp1d

# ===== MODEL PARAMETERS =====
theta = 6.1 * np.pi / 180
m_FP, alpha, beta = 25.0, 0.0, 3.0
Omega_Lambda, Omega_r = 0.699, 9.08e-5
sin2, tan2, m2 = np.sin(theta)**2, np.tan(theta)**2, m_FP**2

# B-parameters (Eq. 3.4, Högås & Mörtsell 2021)
B0 = 3*Omega_Lambda - sin2*m2*(3 + 3*alpha + beta)
B1 = sin2*m2*(1 + 2*alpha + beta)
B2 = -sin2*m2*(alpha + beta)
B3 = sin2*m2*beta
B4 = 3*tan2*Omega_Lambda + sin2*m2*(-1 + alpha - beta)

# ===== QUARTIC SOLVER =====
def solve_y(z, Om):
    Ot = Om*(1+z)**3 + Omega_r*(1+z)**4
    a0 = -B1/(3*tan2)
    a1 = B0/3 - B2/tan2 + Ot
    a2 = B1 - B3/tan2
    a3 = B2 - B4/(3*tan2)
    a4 = B3/3
    return brentq(lambda y: a0 + a1*y + a2*y**2 + a3*y**3 + a4*y**4, 1e-6, 1.0)

def E2_bim(z, Om, y):
    Ot = Om*(1+z)**3 + Omega_r*(1+z)**4
    ODE = Omega_Lambda - sin2*m2*(1-y)*(1 + alpha*(1-y) + beta/3*(1-y)**2)
    return Ot + ODE

def E2_lcdm(z, Om):
    return Om*(1+z)**3 + Omega_r*(1+z)**4 + (1 - Om - Omega_r)

# ===== SELF-CONSISTENCY =====
Omega_m = brentq(lambda Om: E2_bim(0, Om, solve_y(0, Om)) - 1.0, 0.01, 0.99)
y0 = solve_y(0, Omega_m)

# ===== BIMETRIC E(z) ON FINE GRID =====
c_H0 = 2997.92458
z_fine = np.linspace(0, 5, 10000)
y_fine = np.array([solve_y(z, Omega_m) for z in z_fine])
E_fine_bim = np.sqrt(E2_bim(z_fine, Omega_m, y_fine))
int_fine_bim = cumulative_trapezoid(1.0/E_fine_bim, z_fine, initial=0)
int_bim_interp = interp1d(z_fine, int_fine_bim, kind='cubic', fill_value='extrapolate')

# ===== FIND FILES =====
import os
data_file = None
for f in os.listdir('.'):
    fl = f.lower()
    if f.endswith('.dat') and 'sh0es' in fl and 'cov' not in fl:
        data_file = f
        break
print(f"  Data file: {data_file}")
if data_file is None:

    raise FileNotFoundError("Pantheon+SH0ES.dat not found. Download from GitHub (see §3).")

    """LCDM distance modulus for given Omega_m."""
    E = np.sqrt(E2_lcdm(z_fine, Om))
    integ = cumulative_trapezoid(1.0/E, z_fine, initial=0)
    int_z = np.interp(z_arr, z_fine, integ)
    D_L = (1 + zH) * c_H0 * int_z
    return 5 * np.log10(D_L) + 25

# ===== V6 KEY CHANGE: sigma_int in weights =====
# V5: w = 1.0 / err_c**2
# V6: w = 1.0 / (err_c**2 + sigma_int**2)

results = []
for si in sigma_scan:

#!/usr/bin/env python3
"""
CMB chi^2 recalculation for bimetric cosmology.
Compares R formula with and without massive neutrinos.

Based on:
- Bansal & Huterer (2025): CMB shift parameters, data vector, covariance
- Hogas & Mortsell (2025): bimetric quartic, Friedmann equation
- Note: H&M 2025 main text has a typo (y = a_g/a_f); correct is y = a_f/a_g
  (consistent with their Appendix A and H&M 2021)

Key finding:
  Bansal & Huterer INCLUDE neutrinos in R (Eq. 4):
    R = 100 * sqrt(omega_b + omega_cdm + omega_nu) * D_M(z*)/c
  Excluding neutrinos (as in Chernikov) INCREASES chi^2_CMB.

Author: Alisa AI
Date: 2026-10-06
"""

import numpy as np
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')

# ============================================================
# CMB data (Bansal & Huterer 2025, Planck + ACT)
# ============================================================
R_obs   = 1.7504       # shift parameter R
la_obs  = 301.77       # acoustic scale l_a
wb_obs  = 0.022371     # omega_b = Omega_b * h^2
data_vec = np.array([R_obs, la_obs, wb_obs])

# Covariance matrix (x 10^-8)
C_raw = np.array([
    [1559.83,  -1325.41,   -36.45],
    [-1325.41, 714691.80,  269.77],
    [-36.45,   269.77,     2.10]
]) * 1e-8
C_inv = np.linalg.inv(C_raw)

# ============================================================
# Physical constants
# ============================================================
z_star = 1090.0           # recombination redshift
m_nu_sum = 0.06            # sum of neutrino masses (eV)
omega_nu = m_nu_sum / 93.14  # neutrino density ~ 6.44e-4
T_CMB = 2.725               # CMB temperature (K)
omega_gamma_h2 = 2.469e-5    # photon density * h^2
N_eff = 3.046               # effective neutrino species
omega_r_h2 = omega_gamma_h2 * (1 + 0.2271 * N_eff)  # radiation * h^2 ~ 4.178e-5

def R_b_func(wb):
    """Baryon-to-photon ratio R_b = 3*wb / (4*omega_gamma*h^2)."""
    return 3.0 * wb / (4.0 * omega_gamma_h2)

# ============================================================
# E^2(z): expansion rate squared
# Uses LCDM form: Om0*(1+z)^3 + Or*(1+z)^4 + OL
# Normalized so E^2(0) = 1 (handles Om0+OL != 1 in bimetric)
# ============================================================
def E2_of_z(z, Om0, OL, h):
    """E^2(z) normalized to E^2(0) = 1."""
    Or = omega_r_h2 / h**2
    E2_raw = Om0*(1+z)**3 + Or*(1+z)**4 + OL
    E2_0 = Om0 + Or + OL
    return E2_raw / E2_0

# ============================================================
# CMB observables: (R, l_a, omega_b)
# ============================================================
def compute_observables(h, wb, Om0, OL, with_nu=True, wnu=0.0, npts=3000):
    """
    Compute CMB shift parameters.
    
    Parameters:
    -----------
    h : float
        Hubble parameter h = H_0 / 100 km/s/Mpc
    wb : float
        omega_b = Omega_b * h^2
    Om0 : float
        Total matter density Omega_m (includes neutrinos if present)
    OL : float
        Dark energy density Omega_Lambda
    with_nu : bool
        If True: R includes neutrinos (omega_m = Om0*h^2)
        If False: R excludes neutrinos (omega_m = Om0*h^2 - wnu)
    wnu : float
        Neutrino density omega_nu = sum(m_nu)/93.14
    npts : int
        Number of integration points
    
    Returns:
    --------
    np.array([R, l_a, wb])
    """
    # D_M(z*): comoving angular diameter distance / (c/H_0)
    # = integral_0^{z*} dz / E(z)
    z_dm = np.logspace(np.log10(1e-4), np.log10(z_star), npts)
    E_dm = np.sqrt(E2_of_z(z_dm, Om0, OL, h))
    DM = np.trapezoid(1.0 / E_dm, z_dm)
    
    # r_s(z*): comoving sound horizon / (c/H_0)
    # = integral_{z*}^{inf} c_s(z')/E(z') dz'
    # where c_s = 1/sqrt(3*(1 + R_b/(1+z))) in units of c
    z_max = 1e7
    z_rs = np.logspace(np.log10(z_star), np.log10(z_max), npts)
    E_rs = np.sqrt(E2_of_z(z_rs, Om0, OL, h))
    Rb = R_b_func(wb)
    cs_rs = 1.0 / np.sqrt(3.0 * (1.0 + Rb / (1.0 + z_rs)))
    rs = np.trapezoid(cs_rs / E_rs, z_rs)
    
    # R = 100 * sqrt(omega_m) * D_M / c
    # D_M/c = DM / H_0 = DM / (100h)  [in Mpc/(km/s)]
    # R = 100 * sqrt(omega_m) * DM / (100h) = sqrt(omega_m) * DM / h
    if with_nu:
        omega_m = Om0 * h**2           # total matter (incl. neutrinos)
    else:
        omega_m = Om0 * h**2 - wnu     # matter without neutrinos
    R = np.sqrt(omega_m) * DM / h
    
    # l_a = pi * D_M / r_s
    la = np.pi * DM / rs
    
    return np.array([R, la, wb])

def chi2_cmb(model_vec):
    """CMB chi^2 = delta^T C^-1 delta."""
    delta = model_vec - data_vec
    return float(delta @ C_inv @ delta)

# ============================================================
# Minimization over (h, omega_b)
# ============================================================
def minimize_chi2(Om0, OL, with_nu, wnu=0.0):
    """Minimize chi^2 over (h, omega_b). Returns (chi2, h, wb, vec)."""
    def objective(params):
        h, wb = params
        if not (0.6 < h < 0.72 and 0.02 < wb < 0.024):
            return 1e10
        return chi2_cmb(compute_observables(h, wb, Om0, OL, with_nu, wnu))
    
    # Coarse scan for initial guess
    best = 1e10; best_init = None
    for h in np.linspace(0.65, 0.70, 20):
        for wb in np.linspace(0.0215, 0.0230, 20):
            c2 = objective([h, wb])
            if c2 < best:
                best = c2; best_init = [h, wb]
    
    # Nelder-Mead refinement
    res = minimize(objective, best_init, method='Nelder-Mead',
                   options={'xatol': 1e-8, 'fatol': 1e-12, 'maxiter': 5000})
    h_best, wb_best = res.x
    vec = compute_observables(h_best, wb_best, Om0, OL, with_nu, wnu)
    return res.fun, h_best, wb_best, vec

# ============================================================
# MAIN
# ============================================================
if __name__ == '__main__':
    print("=" * 70)
    print("CMB chi^2 Recalculation: Bimetric Cosmology")
    print("With and without massive neutrinos in R")
    print("=" * 70)
    
    print(f"\nData vector (Bansal & Huterer 2025):")
    print(f"  R = {R_obs}, l_a = {la_obs}, omega_b = {wb_obs}")
    print(f"  sum(m_nu) = {m_nu_sum} eV, omega_nu = {omega_nu:.6e}")
    print(f"  z* = {z_star}")
    
    # --- LCDM validation ---
    print("\n--- LCDM validation (Om=0.315, OL=0.685) ---")
    chi2_lcdm, h_lcdm, wb_lcdm, v_lcdm = minimize_chi2(
        0.315, 0.685, with_nu=True, wnu=omega_nu)
    print(f"  h={h_lcdm:.5f}, wb={wb_lcdm:.7f}")
    print(f"  R={v_lcdm[0]:.5f} (obs {R_obs})")
    print(f"  l_a={v_lcdm[1]:.3f} (obs {la_obs})")
    print(f"  chi^2 = {chi2_lcdm:.4f}")
    
    # --- Bimetric model ---
    print("\n" + "=" * 70)
    print("Bimetric model: Omega_m=0.304, Omega_L=0.699")
    print("(Omega_m + Omega_L = 1.003, bimetric correction makes E^2(0)=1)")
    print("=" * 70)
    
    Om_bim = 0.304
    OL_bim = 0.699
    
    # Without neutrinos (Chernikov's formula)
    print("\n--- Without neutrinos (omega_m = Om0*h^2 - omega_nu) ---")
    chi2_no, h_no, wb_no, v_no = minimize_chi2(
        Om_bim, OL_bim, with_nu=False, wnu=omega_nu)
    print(f"  h = {h_no:.5f}")
    print(f"  wb = {wb_no:.7f}")
    print(f"  R = {v_no[0]:.5f} (obs {R_obs})")
    print(f"  l_a = {v_no[1]:.3f} (obs {la_obs})")

def chi2_marginalized_M(mu_data, mu_model, cov_inv):
    r = mu_data - mu_model
    A = np.ones_like(r)
    C_inv_r = cov_inv @ r
    C_inv_A = cov_inv @ A
    M_best = (A @ C_inv_r) / (A @ C_inv_A)
    chi2 = r @ C_inv_r - (A @ C_inv_r)**2 / (A @ C_inv_A)
    return chi2

def compute_sne_dchi2(dat_path, cov_path):
